48    /// Open a single stream if possible
49    ///
50    /// Returns `None` if the streams in the given direction are currently exhausted.
51    pub fn open(&mut self, dir: Dir) -> Option<StreamId> {
52        if self.conn_state.is_closed() {
53            return None;
54        }
55
56        if self.state.next[dir as usize] >= self.state.max[dir as usize] {
57            self.state.streams_blocked[dir as usize] = true;
58            return None;
59        }
60
61        self.state.next[dir as usize] += 1;
62        let id = StreamId::new(self.state.side, dir, self.state.next[dir as usize] - 1);
63        self.state.insert_local(id);
64        self.state.send_streams += 1;
65        Some(id)
66    }
67
68    /// Accept a remotely initiated stream of a certain directionality, if possible
69    ///
70    /// Returns `None` if there are no new incoming streams for this connection.
71    /// Has no impact on the data flow-control or stream concurrency limits.
72    pub fn accept(&mut self, dir: Dir) -> Option<StreamId> {
73        if self.state.next_remote[dir as usize] == self.state.next_reported_remote[dir as usize] {
74            return None;
75        }
76
77        let x = self.state.next_reported_remote[dir as usize];
78        self.state.next_reported_remote[dir as usize] = x + 1;
79        if dir == Dir::Bi {
80            self.state.send_streams += 1;
81        }
82
83        Some(StreamId::new(!self.state.side, dir, x))
84    }
85
86    #[cfg(fuzzing)]
87    pub fn state(&mut self) -> &mut StreamsState {
88        self.state
89    }
90
91    /// The number of streams that may have unacknowledged data.
92    pub fn send_streams(&self) -> usize {
93        self.state.send_streams
94    }
95
96    /// The number of remotely initiated open streams of a certain directionality.
97    ///
98    /// Includes remotely initiated streams, which have not been accepted via
99    /// [`accept`](Self::accept). These streams count against the respective concurrency limit
100    /// reported by
101    /// [`Connection::max_concurrent_streams`](super::Connection::max_concurrent_streams).
102    pub fn remote_open_streams(&self, dir: Dir) -> u64 {
103        // total opened - total closed = total opened - (total permitted - total permitted unclosed)
104        self.state.next_remote[dir as usize]
105            - (self.state.max_remote[dir as usize]
106                - self.state.allocated_remote_count[dir as usize])
107    }
108}

117impl RecvStream<'_> {
118    /// Whether this stream is still in ordered read mode.
119    ///
120    /// A stream switches permanently to unordered mode when [`Self::read`] is called with
121    /// `ordered` set to `false`.
122    pub fn is_ordered(&self) -> Result<bool, ClosedStream> {
123        let Some(stream) = self.state.recv.get(&self.id) else {
124            return Err(ClosedStream { _private: () });
125        };
126        let Some(stream) = stream.as_ref().and_then(StreamRecv::as_open_recv) else {
127            return Ok(true);
128        };
129        if stream.stopped {
130            return Err(ClosedStream { _private: () });
131        }
132        Ok(stream.assembler.is_ordered())
133    }
134
135    /// Read from the given recv stream
136    ///
137    /// `max_length` limits the maximum size of the returned `Bytes` value; passing `usize::MAX`
138    /// will yield the best performance. `ordered` will make sure the returned chunk's offset will
139    /// have an offset exactly equal to the previously returned offset plus the previously returned
140    /// bytes' length.
141    ///
142    /// Yields `Ok(None)` if the stream was finished. Otherwise, yields a segment of data and its
143    /// offset in the stream. If `ordered` is `false`, segments may be received in any order, and
144    /// the `Chunk`'s `offset` field can be used to determine ordering in the caller.
145    ///
146    /// While most applications will prefer to consume stream data in order, unordered reads can
147    /// improve performance when packet loss occurs and data cannot be retransmitted before the flow
148    /// control window is filled. On any given stream, you can switch from ordered to unordered
149    /// reads, but ordered reads on streams that have seen previous unordered reads will return
150    /// `ReadError::IllegalOrderedRead`.
151    pub fn read(&mut self, ordered: bool) -> Result<Chunks<'_>, ReadableError> {
152        Chunks::new(self.id, ordered, self.state, self.pending)
153    }
154
155    /// Stop accepting data on the given receive stream
156    ///
157    /// Discards unread data and notifies the peer to stop transmitting. Once stopped, further
158    /// attempts to operate on a stream will yield `ClosedStream` errors.
159    pub fn stop(&mut self, error_code: VarInt) -> Result<(), ClosedStream> {
160        let mut entry = match self.state.recv.entry(self.id) {
161            hash_map::Entry::Occupied(s) => s,
162            hash_map::Entry::Vacant(_) => return Err(ClosedStream { _private: () }),
163        };
164        let stream = get_or_insert_recv(self.state.stream_receive_window)(entry.get_mut());
165
166        let (read_credits, stop_sending) = stream.stop()?;
167        if stop_sending.should_transmit() {
168            self.pending.stop_sending.push(frame::StopSending {
169                id: self.id,
170                error_code,
171            });
172        }
173
174        // We need to keep stopped streams around until they're finished or reset so we can update
175        // connection-level flow control to account for discarded data. Otherwise, we can discard
176        // state immediately.
177        if !stream.final_offset_unknown() {
178            let recv = entry.remove().expect("must have recv when stopping");
179            self.state.stream_recv_freed(self.id, recv);
180        }
181
182        if self.state.add_read_credits(read_credits).should_transmit() {
183            self.pending.max_data = true;
184        }
185
186        Ok(())
187    }
188
189    /// Returns the number of bytes read from this stream.
190    ///
191    /// This is the offset of the next byte to be read, i.e. the length of the contiguous
192    /// prefix of the stream consumed by the application.
193    pub fn bytes_read(&self) -> Result<u64, ClosedStream> {
194        let recv = self
195            .state
196            .recv
197            .get(&self.id)
198            .and_then(|s| s.as_ref())
199            .and_then(|s| s.as_open_recv())
200            .ok_or(ClosedStream { _private: () })?;
201        Ok(recv.assembler.bytes_read())
202    }
203
204    /// Check whether this stream has been reset by the peer, returning the reset error code if so
205    ///
206    /// After returning `Ok(Some(_))` once, stream state will be discarded and all future calls will
207    /// return `Err(ClosedStream)`.
208    pub fn received_reset(&mut self) -> Result<Option<VarInt>, ClosedStream> {
209        let hash_map::Entry::Occupied(entry) = self.state.recv.entry(self.id) else {
210            return Err(ClosedStream { _private: () });
211        };
212        let Some(s) = entry.get().as_ref().and_then(|s| s.as_open_recv()) else {
213            return Ok(None);
214        };
215        if s.stopped {
216            return Err(ClosedStream { _private: () });
217        }
218        let Some(code) = s.reset_code() else {
219            return Ok(None);
220        };
221
222        // Clean up state after application observes the reset, since there's no reason for the
223        // application to attempt to read or stop the stream once it knows it's reset
224        let (_, recv) = entry.remove_entry();
225        self.state
226            .stream_recv_freed(self.id, recv.expect("must have recv on reset"));
227        self.state.queue_max_stream_id(self.pending);
228
229        Ok(Some(code))
230    }
231}

265    /// Send data on the given stream
266    ///
267    /// Returns the number of bytes written and advances the provided `Bytes`
268    /// slice, removing all completely written chunks.
269    ///
270    /// Note that this method might also write a partial chunk. In this case
271    /// the chunk will be advanced and contain only non-written data after the call.
272    pub fn write_chunks(&mut self, data: &mut &mut [Bytes]) -> Result<usize, WriteError> {
273        let written = self.write_source(&mut BytesArray::from_chunks(data))?;
274        *data = &mut std::mem::take(data)[written.chunks..];
275        Ok(written.bytes)
276    }

278    fn write_source<'b, B: BytesSource<'b>>(
279        &mut self,
280        source: &'b mut B,
281    ) -> Result<Written, WriteError> {
282        if self.conn_state.is_closed() {
283            trace!(%self.id, "write blocked; connection draining");
284            return Err(WriteError::Blocked);
285        }
286
287        let limit = self.state.write_limit();
288
289        let max_send_data = self.state.max_send_data(self.id);
290
291        let stream = self
292            .state
293            .send
294            .get_mut(&self.id)
295            .map(get_or_insert_send(max_send_data))
296            .ok_or(WriteError::ClosedStream)?;
297
298        if limit == 0 {
299            trace!(
300                stream = %self.id, max_data = self.state.max_data, data_sent = self.state.data_sent,
301                "write blocked by connection-level flow control or send window"
302            );
303            if !stream.connection_blocked {
304                stream.connection_blocked = true;
305                self.state.connection_blocked.push(self.id);
306            }
307            return Err(WriteError::Blocked);
308        }
309
310        let was_pending = stream.is_pending();
311        let written = stream.write(source, limit)?;
312        self.state.data_sent += written.bytes as u64;
313        self.state.unacked_data += written.bytes as u64;
314        trace!(stream = %self.id, "wrote {} bytes", written.bytes);
315        if !was_pending {
316            self.state
317                .pending
318                .push_pending(self.id, stream.priority, stream.affinity);
319        }
320        Ok(written)
321    }

332    /// Finish a send stream, signalling that no more data will be sent.
333    ///
334    /// If this fails, no [`StreamEvent::Finished`] will be generated.
335    ///
336    /// [`StreamEvent::Finished`]: crate::StreamEvent::Finished
337    pub fn finish(&mut self) -> Result<(), FinishError> {
338        let max_send_data = self.state.max_send_data(self.id);
339        let stream = self
340            .state
341            .send
342            .get_mut(&self.id)
343            .map(get_or_insert_send(max_send_data))
344            .ok_or(FinishError::ClosedStream)?;
345
346        let was_pending = stream.is_pending();
347        stream.finish()?;
348        if !was_pending {
349            self.state
350                .pending
351                .push_pending(self.id, stream.priority, stream.affinity);
352        }
353
354        Ok(())
355    }

357    /// Abandon transmitting data on a stream
358    ///
359    /// # Panics
360    /// - when applied to a receive stream
361    pub fn reset(&mut self, error_code: VarInt) -> Result<(), ClosedStream> {
362        let max_send_data = self.state.max_send_data(self.id);
363        let stream = self
364            .state
365            .send
366            .get_mut(&self.id)
367            .map(get_or_insert_send(max_send_data))
368            .ok_or(ClosedStream { _private: () })?;
369
370        if matches!(stream.state, SendState::ResetSent) {
371            // Redundant reset call
372            return Err(ClosedStream { _private: () });
373        }
374
375        // Restore the portion of the send window consumed by the data that we aren't about to
376        // send. We leave flow control alone because the peer's responsible for issuing additional
377        // credit based on the final offset communicated in the RESET_STREAM frame we send.
378        self.state.unacked_data -= stream.pending.unacked();
379        stream.reset();
380        self.state.pending.remove(self.id);
381        self.pending.reset_stream.push((self.id, error_code));
382
383        // Don't reopen an already-closed stream we haven't forgotten yet
384        Ok(())
385    }

404    /// Sets the path affinity of a stream.
405    ///
406    /// Applies to pending and future STREAM data, including retransmissions and FINs.
407    /// Does not affect stream control frames.
408    /// Unblocking rejoins the back of the stream's priority group.
409    ///
410    /// # Errors
411    ///
412    /// Returns [`ClosedStream`] if the stream no longer exists.
413    ///
414    /// # Panics
415    ///
416    /// Panics when applied to a receive-only stream.
417    pub fn set_affinity(&mut self, new_affinity: StreamAffinity) -> Result<(), ClosedStream> {
418        let max_send_data = self.state.max_send_data(self.id);
419        let stream = self
420            .state
421            .send
422            .get_mut(&self.id)
423            .map(get_or_insert_send(max_send_data))
424            .ok_or(ClosedStream { _private: () })?;
425
426        if stream.affinity == new_affinity {
427            return Ok(());
428        }
429        if stream.affinity == StreamAffinity::Block {
430            if !stream.is_reset() && stream.is_pending() {
431                self.state
432                    .pending
433                    .push_pending(self.id, stream.priority, new_affinity);
434            }
435        } else {
436            match new_affinity {
437                StreamAffinity::Any => self.state.pending.move_pending(self.id, None),
438                StreamAffinity::Sticky(path) => {
439                    self.state.pending.move_pending(self.id, Some(path))
440                }
441                StreamAffinity::Block => self.state.pending.remove(self.id),
442            }
443        }
444        stream.affinity = new_affinity;
445        Ok(())
446    }

463/// Restricts which paths may carry a stream's STREAM frames.
464#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
465pub enum StreamAffinity {
466    /// Uses the connection's available/backup scheduling policy.
467    #[default]
468    Any,
469    /// Uses only this path, overriding available/backup preference but not transport limits.
470    Sticky(PathId),
471    /// Prevents transmission of STREAM frames for this stream.
472    Block,
473}

475/// Pending streams partitioned by affinity, with shared priority and fairness ordering.
476struct PendingStreamsQueue {
477    any: BinaryHeap<PendingStream>,
478    sticky: IntMap<PathIdKey, BinaryHeap<PendingStream>>,
479    /// Used as a sort of "reverse index" to find in which queue a stream is currently pending.
480    affinities: IntMap<StreamIdKey, Option<PathId>>,
481    /// A monotonically decreasing counter, used to implement round-robin scheduling for streams of
482    /// the same priority. Underflowing is not a practical concern, as it is initialized to
483    /// u64::MAX and only decremented by 1 in `push_pending`
484    recency: u64,
485}
486
487impl PendingStreamsQueue {
488    fn new() -> Self {
489        Self {
490            any: BinaryHeap::new(),
491            sticky: IntMap::default(),
492            affinities: IntMap::default(),
493            recency: u64::MAX,
494        }
495    }
496
497    /// Enqueues a stream in its affinity bucket after queued streams of the same priority.
498    ///
499    /// Ignores blocked streams and preserves any existing queued entry and its turn.
500    fn push_pending(&mut self, id: StreamId, priority: i32, affinity: StreamAffinity) {
501        let sticky = match affinity {
502            StreamAffinity::Any => None,
503            StreamAffinity::Sticky(path) => Some(path),
504            StreamAffinity::Block => return,
505        };
506        // Empty writes and late ACKs can leave an entry queued without pending work.
507        // Re-enqueueing that stream keeps its existing turn rather than duplicating it.
508        match self.affinities.entry(StreamIdKey(id)) {
509            hash_map::Entry::Occupied(_) => return,
510            hash_map::Entry::Vacant(entry) => {
511                entry.insert(sticky);
512            }
513        }
514        // As the recency counter is monotonically decreasing, we know that using its value to sort
515        // this stream will queue it after all other queued streams of the same priority.
516        // This is enough to implement round-robin scheduling for streams that are still pending
517        // even after being handled, as in that case they are removed from the `BinaryHeap`,
518        // handled, and then immediately reinserted.
519        self.recency -= 1;
520        let pending = PendingStream {
521            priority,
522            recency: self.recency,
523            id,
524        };
525        self.queue_mut(sticky).push(pending);
526    }
527
528    /// Removes the highest-ranked entry eligible for this path.
529    ///
530    /// Includes the [`StreamAffinity::Any`] queue only when `include_affinity_any` is true.
531    fn pop(&mut self, path: PathId, include_affinity_any: bool) -> Option<PendingStream> {
532        let any = include_affinity_any.then(|| self.any.peek()).flatten();
533        let sticky = self.sticky.get(&PathIdKey(path)).and_then(|q| q.peek());
534        let pending = if any > sticky {
535            self.queue_mut(None).pop()
536        } else {
537            sticky?; // skip looking at the sticky map, if none exists.
538            self.queue_mut(Some(path)).pop()
539        }?;
540        self.affinities.remove(&StreamIdKey(pending.id));
541        Some(pending)
542    }
543
544    /// Moves a queued stream to the destination queue, preserving priority and recency.
545    ///
546    /// Does nothing if the stream is not queued.
547    fn move_pending(&mut self, id: StreamId, new_affinity: Option<PathId>) {
548        let Some(affinity) = self.affinities.get_mut(&StreamIdKey(id)) else {
549            return;
550        };
551        let old_affinity = std::mem::replace(affinity, new_affinity);
552        let mut moved = None;
553        self.queue_mut(old_affinity).retain(|pending| {
554            if pending.id == id {
555                moved = Some(pending.clone());
556                false
557            } else {
558                true
559            }
560        });
561        self.queue_mut(new_affinity).extend(moved);
562    }
563
564    fn queue_mut(&mut self, sticky: Option<PathId>) -> PendingQueueMut<'_> {
565        match sticky {
566            None => PendingQueueMut::Any(&mut self.any),
567            Some(path) => {
568                let entry = match self.sticky.entry(PathIdKey(path)) {
569                    hash_map::Entry::Occupied(entry) => entry,
570                    hash_map::Entry::Vacant(entry) => entry.insert_entry(BinaryHeap::new()),
571                };
572                PendingQueueMut::Sticky(Some(entry))
573            }
574        }
575    }
576
577    fn remove(&mut self, id: StreamId) {
578        let Some(sticky) = self.affinities.remove(&StreamIdKey(id)) else {
579            return;
580        };
581        // Abandoned-path queues might never be selected again.
582        self.queue_mut(sticky).retain(|pending| pending.id != id);
583    }
584
585    fn clear(&mut self) {
586        self.any.clear();
587        self.sticky.clear();
588        self.affinities.clear();
589    }
590
591    fn iter(
592        &self,
593        path: PathId,
594        include_affinity_any: bool,
595    ) -> impl Iterator<Item = &PendingStream> {
596        include_affinity_any
597            .then_some(&self.any)
598            .into_iter()
599            .flat_map(|q| q.iter())
600            .chain(
601                self.sticky
602                    .get(&PathIdKey(path))
603                    .into_iter()
604                    .flat_map(|q| q.iter()),
605            )
606    }
607
608    #[cfg(test)]
609    fn len(&self) -> usize {
610        self.any.len() + self.sticky.values().map(|q| q.len()).sum::<usize>()
611    }
612}

614/// Helper enum for a mutable reference to a specific queue in the [`PendingStreamsQueue`].
615///
616/// This helper ensures proper cleanup of empty sticky queues in its drop implementation.
617enum PendingQueueMut<'a> {
618    Any(&'a mut BinaryHeap<PendingStream>),
619    // The entry is present until Drop takes ownership to remove an empty bucket.
620    Sticky(Option<hash_map::OccupiedEntry<'a, PathIdKey, BinaryHeap<PendingStream>>>),
621}
622
623impl Deref for PendingQueueMut<'_> {
624    type Target = BinaryHeap<PendingStream>;
625
626    fn deref(&self) -> &Self::Target {
627        match self {
628            Self::Any(queue) => queue,
629            Self::Sticky(entry) => entry.as_ref().expect("entry is present until drop").get(),
630        }
631    }
632}
633
634impl DerefMut for PendingQueueMut<'_> {
635    fn deref_mut(&mut self) -> &mut Self::Target {
636        match self {
637            Self::Any(queue) => queue,
638            Self::Sticky(entry) => entry
639                .as_mut()
640                .expect("entry is present until drop")
641                .get_mut(),
642        }
643    }
644}
645
646impl Drop for PendingQueueMut<'_> {
647    fn drop(&mut self) {
648        if let Self::Sticky(entry) = self {
649            let entry = entry.take().expect("entry is present until drop");
650            if entry.get().is_empty() {
651                entry.remove();
652            }
653        }
654    }
655}

657/// The [`StreamId`] of a stream with pending data queued, ordered by its priority and recency
658#[derive(Clone, PartialEq, Eq, PartialOrd, Ord)]
659struct PendingStream {
660    /// The priority of the stream
661    // Note that this field should be kept above the `recency` field, in order for the `Ord` derive
662    // to be correct (See https://doc.rust-lang.org/stable/std/cmp/trait.Ord.html#derivable)
663    priority: i32,
664    /// A tie-breaker for streams of the same priority, used to improve fairness by implementing
665    /// round-robin scheduling: Larger values are prioritized, so it is initialised to
666    /// `u64::MAX`, and when a stream writes data, we know that it currently has the highest
667    /// recency value, so it is deprioritized by setting its recency to 1 less than the
668    /// previous lowest recency value, such that all other streams of this priority will get
669    /// processed once before we get back round to this one
670    recency: u64,
671    /// The ID of the stream
672    // The way this type is used ensures that every instance has a unique `recency` value, so this
673    // field should be kept below the `priority` and `recency` fields, so that it does not
674    // interfere with the behaviour of the `Ord` derive
675    id: StreamId,
676}

678/// Application events about streams
679#[derive(Debug, PartialEq, Eq)]
680pub enum StreamEvent {
681    /// One or more new streams has been opened and might be readable
682    Opened {
683        /// Directionality for which streams have been opened
684        dir: Dir,
685    },
686    /// A currently open stream likely has data or errors waiting to be read
687    Readable {
688        /// Which stream is now readable
689        id: StreamId,
690    },
691    /// A formerly write-blocked stream might be ready for a write or have been stopped
692    ///
693    /// Only generated for streams that are currently open.
694    Writable {
695        /// Which stream is now writable
696        id: StreamId,
697    },
698    /// A finished stream has been fully acknowledged or stopped
699    Finished {
700        /// Which stream has been finished
701        id: StreamId,
702    },
703    /// The peer asked us to stop sending on an outgoing stream
704    Stopped {
705        /// Which stream has been stopped
706        id: StreamId,
707        /// Error code supplied by the peer
708        error_code: VarInt,
709    },
710    /// At least one new stream of a certain directionality may be opened
711    Available {
712        /// Directionality for which streams are newly available
713        dir: Dir,
714    },
715}

751/// A helper trait to unify Bytes, `Vec<u8>` and `&[u8]` as sources of bytes
752pub(super) trait BytesOrSlice<'a>: AsRef<[u8]> + 'a {
753    fn len(&self) -> usize {
754        self.as_ref().len()
755    }
756    fn is_empty(&self) -> bool {
757        self.as_ref().is_empty()
758    }
759    fn into_bytes(self) -> Bytes;
760}
761
762impl BytesOrSlice<'_> for Bytes {
763    fn into_bytes(self) -> Bytes {
764        self
765    }
766}
767
768impl<'a> BytesOrSlice<'a> for &'a [u8] {
769    fn into_bytes(self) -> Bytes {
770        Bytes::copy_from_slice(self)
771    }
